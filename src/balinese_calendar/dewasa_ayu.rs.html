4//
5// Implementation based on Candana, E.W.H., Gunadi, I.G.A., & Divayana, D.G.H.
6// (2021). "Perbandingan Fuzzy Tsukamoto, Mamdani dan Sugeno dalam Penentuan Hari
7// Baik Pernikahan Berdasarkan Wariga Menggunakan Confusion Matrix". Jurnal Ilmu
8// Komputer Indonesia (JIK), 6(2), 14-22. Universitas Pendidikan Ganesha.
9//
10// Key findings from Candana 2021:
11// - Sugeno FIS: F-1 = 82.76%, Precision = 92.31%, Recall = 75%
12// - Mamdani FIS: F-1 = 5.41%, Precision = 4.76%, Recall = 6.25%
13// - Tsukamoto FIS: F-1 = 4.65%, Precision = 3.70%, Recall = 6.25%
14//
15// Alahaning Dewasa hierarchy (override priority, low → high):
16//   Wewaran → Wuku → Penanggal → Sasih → Dauh
17//
18// Expert ground truth (Pakar Wariga, 2020-2021):
19// - 16/731 days (2.19%) classified as "good" for Pawiwahan
20// - Score-80 days exclusively Buddha or Sukra
21// - Expert NEVER selects Redite or Saniscara
22//
23// Phase 1: Wewaran-only scoring scaffold (superseded)
24// Phase 2: Zero-order Sugeno fuzzy inference engine (current — `DewasaAyu` is
25// wired to it; see `dewasa_ayu_engine` below). `score_saptawara`/
26// `score_pancawara` remain as the Wewaran component feeding `DewasaInput`.
27//
28// LIMITATION (Phase 2): satisfying the <3% full-year rarity gate (see
29// `test_scaffold_rarity_over_full_year`) with no validated Wuku/Penanggal/
30// Sasih prohibition data trades away recall on marginal expert dates — see
31// the `alahaning_dewasa_rules` doc comment for the full rationale. Exact

40/// Configuration for Dewasa Ayu scoring.
41///
42/// Provides configurable thresholds and weights for the classification system.
43/// Default threshold of 0.70 corresponds to Candana's 70-point score threshold.
44#[derive(Debug, Clone, Copy, PartialEq)]
45pub struct DewasaAyuConfig {
46    /// Minimum score (0.0–1.0) to qualify as Dewasa Ayu
47    pub threshold: f64,
48    /// Weight for Saptawara (7-day week) component
49    pub saptawara_weight: f64,
50    /// Weight for Pancawara (5-day week) component
51    pub pancawara_weight: f64,
52}
53
54impl Default for DewasaAyuConfig {
55    fn default() -> Self {
56        Self { threshold: 0.70, saptawara_weight: 0.5, pancawara_weight: 0.5 }
57    }
58}

64    }

69/// Trait for Dewasa Ayu (auspicious day) classification.
70///
71/// Provides methods to compute the auspiciousness score and determine
72/// whether a date qualifies as "Dewasa Ayu" for ceremonies like Pawiwahan.
73///
74/// # Example
75/// ```
76/// use balinese_calendar::{BalineseDate, DewasaAyu};
77///
78/// let date = BalineseDate::from_ymd(2020, 2, 15).unwrap();
79/// let score = date.dewasa_ayu_score();
80/// let is_good = date.is_dewasa_ayu();
81/// ```
82pub trait DewasaAyu {
83    /// Calculate Dewasa Ayu score (0.0–1.0).
84    ///
85    /// Higher scores indicate more auspicious days for ceremonies.
86    /// Based on Wewaran analysis and Candana 2021 expert ground truth.
87    ///
88    /// Score interpretation:
89    /// - 0.80+: Excellent (matches expert score-80 days)
90    /// - 0.70–0.79: Good (matches expert selection threshold)
91    /// - 0.50–0.69: Moderate
92    /// - <0.50: Less favorable
93    fn dewasa_ayu_score(&self) -> f64;
94
95    /// Calculate score with custom configuration.
96    fn dewasa_ayu_score_with_config(&self, config: &DewasaAyuConfig) -> f64;
97
98    /// Check if this date qualifies as Dewasa Ayu (auspicious).
99    ///
100    /// Returns true if the score exceeds the configurable threshold.
101    /// Default threshold is 0.70 (70%), matching Candana 2021.
102    fn is_dewasa_ayu(&self) -> bool;
103
104    /// Check with custom configuration.
105    fn is_dewasa_ayu_with_config(&self, config: &DewasaAyuConfig) -> bool;
106}

112/// Alahaning Dewasa rule base, built once and reused across calls.
113///
114/// `alahaning_dewasa_rules()` allocates a fresh `Vec<SugenoRule>` (fuzzy-set
115/// structs included) on every invocation, which is wasteful on a per-date hot
116/// path — cache it behind a `OnceLock` instead of a rebuild-per-call.
117fn dewasa_ayu_engine() -> &'static sugeno::SugenoEngine {
118    static ENGINE: OnceLock<sugeno::SugenoEngine> = OnceLock::new();
119    ENGINE.get_or_init(sugeno::rule_base::alahaning_dewasa_rules)
120}

122impl DewasaAyu for BalineseDate {
123    fn dewasa_ayu_score(&self) -> f64 {
124        self.dewasa_ayu_score_with_config(&DewasaAyuConfig::default())
125    }
126
127    fn dewasa_ayu_score_with_config(&self, config: &DewasaAyuConfig) -> f64 {
128        // Phase 2: Sugeno fuzzy inference over the Alahaning Dewasa rule base
129        // (task 3.5). Wewaran/Wuku/Penanggal/Sasih/Dauh inputs come from
130        // `DewasaInput`, derived using this date and the caller's weights.
131        let input = sugeno::DewasaInput::from_balinese_date_with_config(self, config);
132        dewasa_ayu_engine().infer(&input).clamp(0.0, 1.0)
133    }
134
135    fn is_dewasa_ayu(&self) -> bool {
136        self.is_dewasa_ayu_with_config(&DewasaAyuConfig::default())
137    }
138
139    fn is_dewasa_ayu_with_config(&self, config: &DewasaAyuConfig) -> bool {
140        self.dewasa_ayu_score_with_config(config) >= config.threshold
141    }
142}

148/// Score Saptawara based on fixture expert distribution.
149///
150/// Actual fixture distribution (library-computed wewaran):
151/// - Wraspati: 12/16 (75%) - highest, appears in all score-80 days
152/// - Sukra: 4/16 (25%) - also appears in score-80 days
153/// - Buda: 0/16 (0%) - favorable but not in this fixture
154/// - Soma: 0/16 (0%)
155/// - Anggara: 0/16 (0%)
156/// - Redite: 0/16 (0%) - never selected
157/// - Saniscara: 0/16 (0%) - never selected
158fn score_saptawara(sapta: &Saptawara) -> f64 {
159    match sapta {
160        Saptawara::Wraspati => 1.0,   // 12 selections (75%), score-80 days
161        Saptawara::Sukra => 0.70,     // 4 selections (25%), score-80 days
162        Saptawara::Buda => 0.40,      // 0 in fixture — no evidence it qualifies
163        Saptawara::Soma => 0.35,      // 0 in fixture
164        Saptawara::Anggara => 0.30,   // 0 in fixture
165        Saptawara::Saniscara => 0.20, // 0 selections (excluded), but highest urip
166        Saptawara::Redite => 0.15,    // 0 selections (excluded)
167    }
168}

238    /// Membership function shape for fuzzy sets.
239    #[derive(Debug, Clone, Copy, PartialEq)]
240    pub enum MembershipShape {
241        /// Triangular: (a, b, c) where b is peak, a and c are base points
242        Triangular { a: f64, b: f64, c: f64 },
243        /// Trapezoidal: (a, b, c, d) where \[b,c\] is plateau, a and d are outer points
244        Trapezoidal { a: f64, b: f64, c: f64, d: f64 },
245    }
246
247    /// Fuzzy set with a membership function.
248    #[derive(Debug, Clone, Copy)]
249    pub struct FuzzySet {
250        pub shape: MembershipShape,
251        pub linguistic: LinguisticValue,
252    }
253
254    impl FuzzySet {
255        /// Create a triangular fuzzy set.
256        pub fn triangular(a: f64, b: f64, c: f64, linguistic: LinguisticValue) -> Self {
257            Self { shape: MembershipShape::Triangular { a, b, c }, linguistic }
258        }
259
260        /// Create a trapezoidal fuzzy set.
261        pub fn trapezoidal(a: f64, b: f64, c: f64, d: f64, linguistic: LinguisticValue) -> Self {
262            Self { shape: MembershipShape::Trapezoidal { a, b, c, d }, linguistic }
263        }
264
265        /// Compute membership degree (μ) for a crisp input value.
266        ///
267        /// Returns value in [0.0, 1.0] range.
268        pub fn membership(&self, x: f64) -> f64 {
269            let x = x.clamp(0.0, 1.0);
270            match self.shape {
271                MembershipShape::Triangular { a, b, c } => {
272                    if x <= a || x >= c {
273                        0.0
274                    } else if x == b {
275                        1.0
276                    } else if x < b {
277                        (x - a) / (b - a)
278                    } else {
279                        (c - x) / (c - b)
280                    }
281                }
282                MembershipShape::Trapezoidal { a, b, c, d } => {
283                    if x <= a || x >= d {
284                        0.0
285                    } else if x >= b && x <= c {
286                        1.0
287                    } else if x < b {
288                        (x - a) / (b - a)
289                    } else {
290                        (d - x) / (d - c)
291                    }
292                }
293            }
294        }
295    }

314    impl DewasaInput {
315        /// Create input from individual normalized scores.
316        pub fn new(wewaran: f64, wuku: f64, penanggal: f64, sasih: f64, ala_ayu: f64) -> Self {
317            Self {
318                wewaran: wewaran.clamp(0.0, 1.0),
319                wuku: wuku.clamp(0.0, 1.0),
320                penanggal: penanggal.clamp(0.0, 1.0),

326        /// Derive Sugeno input variables from a BalineseDate with custom weights.
327        ///
328        /// Normalizes all five input variables to [0.0, 1.0] for fuzzy inference:
329        /// - **wewaran**: Weighted composite of Saptawara + Pancawara scores (uses config weights)
330        /// - **wuku**: Wuku position (0–29 index) normalized by dividing by 29 to reach [0.0, 1.0]
331        /// - **penanggal**: Tithi (lunar day, 1–30) normalized by dividing by 30
332        /// - **sasih**: Sasih month (0–13 enum value) normalized by dividing by 13
333        /// - **ala_ayu**: Base auspiciousness (estimated from Wewaran for Phase 2; pending Ariana & Budayoga bobot tables)
334        pub fn from_balinese_date_with_config(
335            date: &BalineseDate,
336            config: &DewasaAyuConfig,
337        ) -> Self {
338            // Wewaran: weighted combination using config weights
339            let sapta_score = score_saptawara(&date.saptawara);
340            let panca_score = score_pancawara(&date.pancawara);
341            let wewaran =
342                config.saptawara_weight * sapta_score + config.pancawara_weight * panca_score;
343
344            // Wuku: position within 30-wuku cycle (0–29), normalized to [0, 1]
345            let wuku_norm = (date.wuku.index() as f64) / 29.0;
346
347            // Penanggal: tithi (1–30), normalized to [0, 1]
348            let tithi = date.sasih_day.tithi_number() as f64;
349            let penanggal_norm = tithi / 30.0;
350
351            // Sasih: month (0–13 enum value), normalized to [0, 1]
352            // Enum values: 0–11 for regular months, 12–13 for intercalary
353            let sasih_idx = date.sasih as u32 as f64;
354            let sasih_norm = sasih_idx / 13.0;
355
356            // Ala-Ayu: placeholder, use wewaran as estimate (Phase 2: will derive from Ariana & Budayoga bobot)
357            let ala_ayu_est = wewaran;
358
359            Self::new(wewaran, wuku_norm, penanggal_norm, sasih_norm, ala_ayu_est)
360        }
361
362        /// Derive Sugeno input variables from a BalineseDate using default config weights.
363        ///
364        /// Convenience method that uses `DewasaAyuConfig::default()` weights.
365        /// For custom weights, use `from_balinese_date_with_config()`.
366        pub fn from_balinese_date(date: &BalineseDate) -> Self {
367            Self::from_balinese_date_with_config(date, &DewasaAyuConfig::default())
368        }
369    }

371    /// Zero-order Sugeno fuzzy rule with 5 antecedents and constant consequent.
372    #[derive(Debug, Clone)]
373    pub struct SugenoRule {
374        /// Antecedent fuzzy sets for each input variable
375        pub wewaran_set: FuzzySet,
376        pub wuku_set: FuzzySet,
377        pub penanggal_set: FuzzySet,
378        pub sasih_set: FuzzySet,
379        pub ala_ayu_set: FuzzySet,
380        /// Consequent: constant output value (typically LinguisticValue center)
381        pub output: f64,
382    }
383
384    impl SugenoRule {
385        /// Create a new rule with specified fuzzy sets and output.
386        pub fn new(
387            wewaran_set: FuzzySet,
388            wuku_set: FuzzySet,
389            penanggal_set: FuzzySet,
390            sasih_set: FuzzySet,
391            ala_ayu_set: FuzzySet,
392            output: f64,
393        ) -> Self {
394            Self { wewaran_set, wuku_set, penanggal_set, sasih_set, ala_ayu_set, output }
395        }
396
397        /// Compute firing strength using product t-norm.
398        ///
399        /// Formula: μ_rule = μ_wewaran × μ_wuku × μ_penanggal × μ_sasih × μ_ala_ayu
400        pub fn firing_strength(&self, input: &DewasaInput) -> f64 {
401            let mu_w = self.wewaran_set.membership(input.wewaran);
402            let mu_u = self.wuku_set.membership(input.wuku);
403            let mu_p = self.penanggal_set.membership(input.penanggal);
404            let mu_s = self.sasih_set.membership(input.sasih);
405            let mu_a = self.ala_ayu_set.membership(input.ala_ayu);
406
407            // Product t-norm (AND operation)
408            mu_w * mu_u * mu_p * mu_s * mu_a
409        }
410    }

412    /// Zero-order Sugeno Fuzzy Inference Engine.
413    ///
414    /// Implements weighted average defuzzification for multiple rules.
415    #[derive(Debug, Clone, Default)]
416    pub struct SugenoEngine {
417        pub rules: Vec<SugenoRule>,
418    }
419
420    impl SugenoEngine {
421        /// Create empty engine.
422        pub fn new() -> Self {
423            Self { rules: Vec::new() }
424        }
425
426        /// Create engine with predefined rules.
427        pub fn with_rules(rules: Vec<SugenoRule>) -> Self {
428            Self { rules }
429        }
430
431        /// Add a rule to the engine.
432        pub fn add_rule(&mut self, rule: SugenoRule) {
433            self.rules.push(rule);
434        }
435
436        /// Run fuzzy inference with weighted average defuzzification.
437        ///
438        /// Formula: output = Σ(μi × zi) / Σ(μi)
439        /// where μi = firing strength of rule i, zi = consequent constant
440        ///
441        /// Returns value in [0.0, 1.0] range, or 0.5 if no rules fire.
442        pub fn infer(&self, input: &DewasaInput) -> f64 {
443            if self.rules.is_empty() {
444                return 0.5; // Neutral when no rules defined
445            }
446
447            let fired: Vec<(f64, f64)> = self
448                .rules
449                .iter()
450                .map(|r| (r.firing_strength(input), r.output))
451                .filter(|(strength, _)| *strength > 0.0)
452                .collect();
453
454            if fired.is_empty() {
455                return 0.0; // No rules fired
456            }
457
458            let numerator: f64 = fired.iter().map(|(w, z)| w * z).sum();
459            let denominator: f64 = fired.iter().map(|(w, _)| *w).sum();
460
461            if denominator == 0.0 { 0.0 } else { (numerator / denominator).clamp(0.0, 1.0) }
462        }
463
464        /// Check if inference result qualifies as Dewasa Ayu (auspicious).
465        pub fn is_auspicious(&self, input: &DewasaInput, threshold: f64) -> bool {
466            self.infer(input) >= threshold.clamp(0.0, 1.0)
467        }
468    }

499    /// The Alahaning Dewasa rule base.
500    ///
501    /// Hand-authored conjunctive (product t-norm) Sugeno rules ordered by the
502    /// traditional override hierarchy (low → high): Wewaran → Wuku → Penanggal
503    /// → Sasih → Dauh. `DewasaInput::ala_ayu` stands in for Dauh until the
504    /// Ariana & Budayoga bobot tables are available (see module docs).
505    ///
506    /// The hierarchy is encoded as *veto strength*: a variable higher in the
507    /// chain can override an otherwise-excellent Wewaran, and the higher its
508    /// rank, the harder it drags the output down when its value is bad. This
509    /// gives the literal DoD requirement — a "good" output needs high Wewaran
510    /// **and** acceptable Wuku/Penanggal/Sasih — without special-cased
511    /// branching: the "auspicious" rules simply have zero membership (and so
512    /// don't fire) once one of those falls into weak/prohibited territory,
513    /// leaving only the override rules to determine the (low) blended output.
514    ///
515    /// Wuku, Penanggal, and Sasih have no validated prohibition data yet (the
516    /// Ariana & Budayoga bobot tables are pending — see module docs), and
517    /// `verify_finding_experts_span_space` (tests/dewasa_ayu_test.rs) shows the
518    /// 16 Candana 2021 expert dates occupy nearly the *entire* Wuku/Penanggal/
519    /// Sasih range while Wewaran stays tightly clustered at 0.875–1.0. That
520    /// spread makes the two DoDs — a full-year positive rate under 3% (the
521    /// hard gate; see `test_scaffold_rarity_over_full_year`) and 16/16 expert
522    /// recall — mutually exclusive with only a marginal "exclude the deep low
523    /// tail" band on Wuku/Penanggal/Sasih (empirically that band alone leaves
524    /// ~94% of days classified "good", nowhere near 3%). Since Plans.md scopes
525    /// rarity as the hard gate and exact metric replication as best-effort,
526    /// `moderate_or_better` below trades some recall for rarity: Wuku,
527    /// Penanggal, and Sasih must each clear a real (if still provisional)
528    /// floor, not just avoid the extreme tail. This keeps the 3 expert dates
529    /// where all three are comfortably clear of that floor, and lets the
530    /// override rules (branch C) continue to handle genuinely bad values.
531    /// Precise prohibition calibration remains best-effort until the bobot
532    /// tables land (tracked by task 3.7 in Plans.md).
533    pub mod rule_base {
534        use super::*;
535
536        /// Build the populated `SugenoEngine` for Dewasa Ayu (Pawiwahan) classification.
537        pub fn alahaning_dewasa_rules() -> SugenoEngine {
538            let sets = standard_sets::triangular_five();
539            let sbr = sets[0];
540            let br = sets[1];
541            let s = sets[2];
542            let b = sets[3];
543            // `standard_sets::triangular_five()`'s SB is a symmetric triangle
544            // that returns membership 0 exactly at x=1.0 — the wrong shape
545            // for an extremal "highest" category. Real Wewaran/ala_ayu
546            // values commonly land exactly at 1.0 (e.g. Wraspati paired with
547            // Wage or Kliwon), so use a proper right-shoulder trapezoid
548            // (plateau from 0.9 through 1.0) scoped to this rule base.
549            let sb = FuzzySet::trapezoidal(0.75, 0.9, 1.0, 1.05, LinguisticValue::SB);
550
551            // Provisional "acceptable" floor for Wuku/Penanggal/Sasih — see the
552            // module-level doc comment above for why this can't yet be a
553            // validated prohibition band, and why it must be tighter than a
554            // "just avoid the extreme tail" band to satisfy the rarity gate.
555            let moderate_or_better =
556                FuzzySet::trapezoidal(0.50, 0.65, 1.0, 1.05, LinguisticValue::B);
557
558            let mut engine = SugenoEngine::new();
559
560            // A. Auspicious branch — requires high Wewaran AND acceptable
561            // Wuku/Penanggal/Sasih.
562            engine.add_rule(SugenoRule::new(sb, sb, sb, sb, sb, 0.95)); // excellent: everything aligned
563            engine.add_rule(SugenoRule::new(
564                sb,
565                moderate_or_better,
566                moderate_or_better,
567                moderate_or_better,
568                sb,
569                0.80,
570            )); // good (SB tier)
571
572            // B. Moderate branch — all signals agree but none are extreme.
573            engine.add_rule(SugenoRule::new(s, s, s, s, s, 0.50));
574            engine.add_rule(SugenoRule::new(b, s, s, s, s, 0.55));
575
576            // C. Override branch — a bad value on a higher-priority variable
577            // vetoes an otherwise-excellent Wewaran. Damage increases up the
578            // hierarchy: Penanggal < Sasih < Dauh (ala_ayu).
579            engine.add_rule(SugenoRule::new(sb, sb, sbr, s, s, 0.15));
580            engine.add_rule(SugenoRule::new(sb, sb, s, sbr, s, 0.10));
581            engine.add_rule(SugenoRule::new(sb, sb, s, s, sbr, 0.05));
582
583            // D. Weak-Wewaran branch — the base signal alone cannot carry a
584            // good day, regardless of the higher-priority variables.
585            engine.add_rule(SugenoRule::new(sbr, s, s, s, s, 0.20));
586            engine.add_rule(SugenoRule::new(br, s, s, s, s, 0.30));
587
588            engine
589        }
590    }
591}
